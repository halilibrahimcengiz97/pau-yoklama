                    📧 E-posta Ayarları

                    Google Apps Script ile ücretsiz, sınırsız mail gönderin. Kurulum için aşağıya tıklayın. Script URL'sini bir kez girin — sistem her oturum kapanışında otomatik mail atar.

                    ⚙️ Google Apps Script Kurulumu

                    Aşağıdaki adımları bir kez yapın (5 dakika):

                    1. script.google.com → Yeni Proje
                    2. Aşağıdaki kodu yapıştırın ve kaydedin
                    3. Dağıt → Web uygulaması olarak dağıt → Erişim: Herkes
                    4. Oluşan URL'yi yukarıya yapıştırın
                    function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var recipients = data.recipients || [];
    var subject = data.subject || 'Yoklama Raporu';
    var body = data.body || '';
    var rows = data.rows || [];
    var sheetName = data.sheetName || 'Yoklama';

    // Excel (CSV) eki oluştur
    var attachments = [];
    if (rows.length > 0) {
      var csv = '﻿'; // UTF-8 BOM (Excel için)
      csv += 'Sıra,Öğrenci No,Ad Soyad,Giriş Saati\n';
      rows.forEach(function(r) {
        csv += r.sira + ',' + r.no + ',"' + r.ad + '",' + r.saat + '\n';
      });
      var blob = Utilities.newBlob(csv, 'text/csv; charset=utf-8', sheetName + '.csv');
      attachments.push(blob);
    }

    recipients.forEach(function(email) {
      try {
        GmailApp.sendEmail(email, subject, body, {
          attachments: attachments,
          name: 'PAÜ Tıp Fakültesi Yoklama Sistemi'
        });
      } catch(err) {}
    });

    return ContentService
      .createTextOutput(JSON.stringify({ok: true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok: false, error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
